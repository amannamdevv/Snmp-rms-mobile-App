// ─── API call ─────────────────────────────────────────────────────────────────
async function loadData(section = currentTab, page = 1, repPage = 1) {
    try {
        const url = new URL(API_URL, window.location.origin);
        url.searchParams.set('section', section);
        url.searchParams.set('page', page);
        url.searchParams.set('rep_page', repPage);

        const res = await fetch(url.toString(), { credentials: 'same-origin' });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        apiData = await res.json();
        renderAll(apiData);
    } catch (err) {
        showToast('Failed to load data: ' + err.message, 'error');
        setLoadingError();
    }
}

// ─── Render all sections from response ───────────────────────────────────────
function renderAll(data) {
    if (data.equipment) {
        renderEquipmentTab(data.equipment);
    }
    if (data.tickets) {
        renderTickets(data.tickets);
    }
    if (data.major_repairs) {
        renderRepairs(data.major_repairs);
    }
}

        tbody.innerHTML = `<tr><td colspan="6" style="text-align:center;padding:20px;color:#999;">No data available</td></tr>`;

function renderEquipmentPagination(pagination) {

    container.innerHTML = html;
}

function switchTab(tabName) {
    document.querySelectorAll('.tab-trigger[data-tab]').forEach(b => b.classList.remove('active'));
    document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));

    const btn = document.querySelector(`.tab-trigger[data-tab="${tabName}"]`);
    const content = document.getElementById(tabName);
    if (btn) btn.classList.add('active');
    if (content) content.classList.add('active');

    currentTab = tabName;
    const url = new URL(window.location);
    url.searchParams.set('section', tabName);
    window.history.replaceState({}, '', url);

    // If data not yet loaded, load it now

}

// ─── Helpers ──────────────────────────────────────────────────────────────────
function esc(str) {
    if (str == null) return '';
    return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function capitalize(str) {
    if (!str) return '-';
    return String(str).charAt(0).toUpperCase() + String(str).slice(1);
}

function setText(id, val) {
    const el = document.getElementById(id);
    if (el) el.textContent = val;
}

    });
}

// ─── Sort ─────────────────────────────────────────────────────────────────────
function sortTable(th, columnIndex) {
    const table = th.closest('table');
    const tbody = table.querySelector('tbody');
    const rows = Array.from(tbody.querySelectorAll('tr'));
    const isAscending = th.classList.contains('sort-asc');
    table.querySelectorAll('th').forEach(h => h.classList.remove('sort-asc', 'sort-desc'));
    th.classList.add(isAscending ? 'sort-desc' : 'sort-asc');
    rows.sort((a, b) => {
        const av = a.cells[columnIndex]?.textContent.trim() || '';
        const bv = b.cells[columnIndex]?.textContent.trim() || '';
        return isAscending ? bv.localeCompare(av) : av.localeCompare(bv);
    });
    rows.forEach(row => tbody.appendChild(row));
}
